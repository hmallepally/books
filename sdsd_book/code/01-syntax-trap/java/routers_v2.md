import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.web.servlet.MockMvc;
import org.springframework.test.web.servlet.MvcResult;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.assertj.core.api.Assertions.assertThat;
@SpringBootTest
public class InvariantCachingTest {
    @Autowired
    private MockMvc mockMvc;
    @Test
    public void testInvariantStrictTenantIsolation() throws Exception {
        /**
         * INVARIANT: It must be mathematically impossible for User B 
         * to retrieve a cache payload generated by User A.
         */
        // 1. User A generates a cache payload
        MvcResult resA = mockMvc.perform(get("/api/v1/transactions?limit=10")
                .header("Authorization", "Bearer token-user-A"))
                .andReturn();
        assertThat(resA.getResponse().getStatus()).isEqualTo(200);
        // 2. User B executes the exact same query parameters
        MvcResult resB = mockMvc.perform(get("/api/v1/transactions?limit=10")
                .header("Authorization", "Bearer token-user-B"))
                .andReturn();
        assertThat(resB.getResponse().getStatus()).isEqualTo(200);
        // 3. VERIFY INVARIANT: The data must differ. User B must not see User A's data.
        // If the cache is insecurely hashing just the query string, this assertion will fail.
        String jsonA = resA.getResponse().getContentAsString();
        String jsonB = resB.getResponse().getContentAsString();
        assertThat(jsonA).as("INVARIANT BREACH: Cross-tenant data exposure detected.")
            .isNotEqualTo(jsonB);
    }
}
