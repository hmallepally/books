import pytest
from httpx import AsyncClient
@pytest.mark.asyncio
async def test_invariant_strict_tenant_isolation(test_app, mock_redis, auth_user_a, auth_user_b):
    """
    INVARIANT: It must be mathematically impossible for User B 
    to retrieve a cache payload generated by User A.
    """
    # 1. User A generates a cache payload
    res_a = await test_app.get(
        "/api/v1/transactions?limit=10", 
        headers=auth_user_a.headers
    )
    assert res_a.status_code == 200
    # 2. User B executes the exact same query parameters
    res_b = await test_app.get(
        "/api/v1/transactions?limit=10", 
        headers=auth_user_b.headers
    )
    assert res_b.status_code == 200
    # 3. VERIFY INVARIANT: The data must differ. User B must not see User A's data.
    # If the cache is insecurely hashing just the query string, this assertion will fail.
    assert res_a.json() != res_b.json(), "INVARIANT BREACH: Cross-tenant data exposure detected."
